五、核心调度代码（可直接用）

你可以新建一个：

ai_router.py
✅ 完整示例代码
import requests
import os

BASE_URL = os.getenv("OPENAI_BASE_URL")
API_KEY = os.getenv("OPENAI_API_KEY")

def call_model(model, prompt):
    url = f"{BASE_URL}/chat/completions"
    headers = {
        "Authorization": f"Bearer {API_KEY}"
    }

    data = {
        "model": model,
        "messages": [
            {"role": "user", "content": prompt}
        ]
    }

    try:
        res = requests.post(url, headers=headers, json=data, timeout=30)
        return res.json()["choices"][0]["message"]["content"]
    except:
        return None


def route(task_type, prompt):
    if task_type == "analysis":
        model = os.getenv("MODEL_ANALYSIS")

    elif task_type == "longtext":
        model = os.getenv("MODEL_LONGTEXT")

    elif task_type == "fast":
        model = os.getenv("MODEL_FAST")

    else:
        model = os.getenv("MODEL_MAIN")

    result = call_model(model, prompt)

    # 兜底机制
    if result is None:
        result = call_model(os.getenv("MODEL_FALLBACK"), prompt)

    return result
